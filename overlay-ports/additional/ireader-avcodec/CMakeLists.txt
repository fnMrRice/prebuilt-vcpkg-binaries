cmake_minimum_required(VERSION 3.29)
project(ireader-avcodec VERSION 1.0.0 LANGUAGES C CXX)
set(IREADER_INCLUDE_PREFIX ireader/avcodec)
include(cmake/library.cmake)
foreach(module IN ITEMS h264 h265 avcodec avbsf)
    string(TOUPPER "${module}" upper)
    option(BUILD_LIBRARY_${upper} "Build ${module}" ON)
endforeach()
# vcpkg feature options use uppercase names.
if(BUILD_LIBRARY_H264 OR BUILD_LIBRARY_H265 OR BUILD_LIBRARY_AVBSF)
    ireader_library(h264 "${CMAKE_CURRENT_SOURCE_DIR}/h264/include" h264 "${CMAKE_CURRENT_SOURCE_DIR}/h264/source")
endif()
if(BUILD_LIBRARY_H265 OR BUILD_LIBRARY_AVBSF)
    ireader_library(h265 "${CMAKE_CURRENT_SOURCE_DIR}/h265/include" h265 "${CMAKE_CURRENT_SOURCE_DIR}/h265/src")
    target_link_libraries(h265 PUBLIC h264)
endif()
if(BUILD_LIBRARY_AVCODEC OR BUILD_LIBRARY_AVBSF)
    find_package(ireader-sdk CONFIG REQUIRED COMPONENTS sdk)
    ireader_library(avcodec "${CMAKE_CURRENT_SOURCE_DIR}/avcodec/include" avcodec "${CMAKE_CURRENT_SOURCE_DIR}/avcodec/source")
    get_target_property(codec_sources avcodec SOURCES)
    list(FILTER codec_sources EXCLUDE REGEX [[/(audio-resampler|faac-decoder|faac-encoder|fdk-decoder|mp2lame-encoder|openh264enc|opus-decoder|opus-encoder|text-render-freetype|x262enc|x264enc)\.(c|cpp)$]])
    set_property(TARGET avcodec PROPERTY SOURCES "${codec_sources}")
    # Avoid colliding with FFmpeg's avcodec library in the same install tree.
    set_target_properties(avcodec PROPERTIES OUTPUT_NAME ireader-avcodec)
    target_link_libraries(avcodec PUBLIC ireader::sdk::sdk)
    if(WIN32)
        target_link_libraries(avcodec PUBLIC gdi32 user32)
    else()
        target_link_libraries(avcodec PUBLIC m)
    endif()
endif()
if(BUILD_LIBRARY_AVBSF)
    find_package(ireader-media-common CONFIG REQUIRED)
    ireader_library(avbsf "${CMAKE_CURRENT_SOURCE_DIR}/avbsf/include" avbsf "${CMAKE_CURRENT_SOURCE_DIR}/avbsf/src")
    target_compile_definitions(avbsf PUBLIC __ERROR__=0)
    target_link_libraries(avbsf PUBLIC avcodec h264 h265 ireader::media-server::flv ireader::media-server::metadata)
endif()
include(cmake/package.cmake)
